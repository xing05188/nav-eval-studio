"""码极客具身导航多模态流式评测与诊断平台 V1.0 启动入口。

用法：
    python run.py                     # 默认 127.0.0.1:8080
    python run.py --port 9000
    python run.py --no-browser --reload
"""

from __future__ import annotations

import argparse
import os
import sys
import threading
import time
import webbrowser
from pathlib import Path

ROOT_DIR = Path(__file__).resolve().parent
if str(ROOT_DIR) not in sys.path:
    sys.path.insert(0, str(ROOT_DIR))


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(description="码极客具身导航多模态流式评测与诊断平台 V1.0")
    parser.add_argument("--host", default=None, help="监听地址，默认取配置文件")
    parser.add_argument("--port", type=int, default=None, help="监听端口，默认取配置文件")
    parser.add_argument("--config", default=None, help="配置文件路径")
    parser.add_argument("--no-browser", action="store_true", help="启动后不自动打开浏览器")
    parser.add_argument("--reload", action="store_true", help="开发模式，代码变更自动重载")
    return parser.parse_args()


def open_browser_later(url: str, delay: float = 1.5) -> None:
    def _open() -> None:
        time.sleep(delay)
        try:
            webbrowser.open(url)
        except Exception:  # noqa: BLE001 - 无图形环境时忽略
            pass

    threading.Thread(target=_open, daemon=True).start()


def main() -> None:
    args = parse_args()
    if args.config:
        os.environ["NAVEVAL_CONFIG"] = str(Path(args.config).resolve())

    # 读取配置需要先导入；bootstrap 在模块导入时完成
    from backend.app.core.config import load_settings, set_settings

    settings = load_settings(args.config)
    set_settings(settings)

    host = args.host or settings.server.host
    port = args.port or settings.server.port
    url = f"http://{'127.0.0.1' if host in ('0.0.0.0', '::') else host}:{port}"

    if not args.no_browser and settings.server.open_browser and not args.reload:
        open_browser_later(url)

    print("=" * 64)
    print(" 码极客具身导航多模态流式评测与诊断平台 V1.0")
    print(f" 访问地址：{url}")
    print(f" 接口文档：{url}/docs")
    print(f" 工作目录：{settings.storage.work_dir}")
    print("=" * 64)

    import uvicorn

    uvicorn.run(
        "backend.app.main:app",
        host=host,
        port=port,
        reload=args.reload,
        log_level=settings.logging.level.lower(),
        # 默认 5 s 太短：页面静一会儿后浏览器复用刚被服务端关掉的连接，
        # POST /api/sim/actions 会报 ERR_INVALID_HTTP_RESPONSE（浏览器不会重试 POST）
        timeout_keep_alive=30,
    )


if __name__ == "__main__":
    main()

def bootstrap() -> None:
    """初始化配置、日志、密钥与数据库，并写入种子数据。"""
    settings = get_settings()
    setup_logging(settings)
    init_secret(settings.secret_file)
    set_iterations(settings.security.pbkdf2_iterations)
    first_time = init_database(settings)
    with session_scope() as session:
        summary = seed_all(session, settings)
    logger.info(
        "初始化完成 first_time=%s 种子=%s 工作目录=%s",
        first_time,
        summary,
        settings.storage.work_dir,
    )


@asynccontextmanager
async def lifespan(app: FastAPI):
    BUS.bind_loop(asyncio.get_running_loop())
    scheduler = get_scheduler()
    await scheduler.start()
    app.state.scheduler = scheduler
    auto_importer = get_auto_importer()
    await auto_importer.start()
    app.state.backup_auto_importer = auto_importer
    logger.info("服务已启动")
    yield
    await auto_importer.shutdown()
    await scheduler.shutdown()
    # 仿真线程持有可能残留的 headless GL context，必须在进程退出前串行释放
    await asyncio.to_thread(sim_session.shutdown)
    logger.info("服务已停止")


def create_app() -> FastAPI:
    bootstrap()

    app = FastAPI(
        title="码极客具身导航多模态流式评测与诊断平台",
        version="1.0.0",
        description="码极客 NavEval Studio 服务端接口",
        lifespan=lifespan,
    )

    app.add_middleware(
        CORSMiddleware,
        allow_origins=["http://localhost:5173", "http://127.0.0.1:5173"],
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )

    for module in (
        auth,
        users,
        system,
        overview,
        audit,
        datasets,
        splits,
        imports,
        models,
        templates,
        tasks,
        analysis,
        reports,
        replay,
        sim,
    ):
        app.include_router(module.router, prefix="/api")

    @app.get("/health")
    def health() -> dict:
        return {"status": "ok", "version": "1.0.0"}

    @app.exception_handler(AppError)
    async def handle_app_error(_: Request, exc: AppError) -> JSONResponse:
        return JSONResponse(
            status_code=exc.http_status,
            content={"code": exc.code, "message": exc.message, "data": None},
        )

    @app.exception_handler(RequestValidationError)
    async def handle_validation_error(_: Request, exc: RequestValidationError) -> JSONResponse:
        first = exc.errors()[0] if exc.errors() else {}
        field = ".".join(str(part) for part in first.get("loc", [])[1:]) or "参数"
        return JSONResponse(
            status_code=400,
            content={
                "code": PARAM_INVALID,
                "message": f"{field}: {first.get('msg', '参数不合法')}",
                "data": {"errors": exc.errors()},
            },
        )

    @app.exception_handler(StarletteHTTPException)
    async def handle_http_error(_: Request, exc: StarletteHTTPException) -> JSONResponse:
        code = UNAUTHORIZED if exc.status_code == 401 else NOT_FOUND
        return JSONResponse(
            status_code=exc.status_code,
            content={"code": code if exc.status_code < 500 else INTERNAL_ERROR,
                     "message": str(exc.detail), "data": None},
        )

    @app.exception_handler(Exception)
    async def handle_unexpected(request: Request, exc: Exception) -> JSONResponse:
        logger.exception("未处理异常 %s %s", request.method, request.url.path)
        return JSONResponse(
            status_code=500,
            content={"code": INTERNAL_ERROR, "message": "服务内部错误", "data": None},
        )

    _mount_frontend(app)
    return app


def _mount_frontend(app: FastAPI) -> None:
    index_file = FRONTEND_DIST / "index.html"
    assets_dir = FRONTEND_DIST / "assets"

    if assets_dir.exists():
        app.mount("/assets", StaticFiles(directory=assets_dir), name="assets")

    if index_file.exists():

        @app.get("/{full_path:path}", include_in_schema=False)
        async def spa(full_path: str):
            candidate = (FRONTEND_DIST / full_path).resolve()
            if (
                full_path
                and FRONTEND_DIST.resolve() in candidate.parents
                and candidate.is_file()
            ):
                return FileResponse(candidate)
            # 入口 HTML 必须不缓存：否则前端重新构建后，浏览器仍用旧 HTML 去加载已删除的 chunk
            return FileResponse(
                index_file,
                headers={"Cache-Control": "no-cache, no-store, must-revalidate"},
            )

    else:

        @app.get("/", include_in_schema=False)
        async def placeholder():
            return HTMLResponse(PLACEHOLDER_HTML)


app = create_app()

@dataclass
class ServerConfig:
    host: str = "0.0.0.0"

@dataclass
class StorageConfig:

@dataclass

@dataclass
class ImportConfig:
    """历史评测产物与备份库自动导入的默认绑定。"""

    backup_root: Path = ROOT_DIR / "storage" / "sample_results"

    auto_interval_s: int = 300

    running_grace_s: int = 600


@dataclass
class SimConfig:
    """交互式仿真：场景根目录与渲染/会话参数（全部有默认值，缺省也能启动）。"""

    scene_root: Path | None = None
    gpu_ids: list[int] = field(default_factory=lambda: [0])
    tick_hz: float = 4.0
    render_hz: float = 15.0
    stream_hz: float = 20.0
    jpeg_quality: int = 82
    max_clients: int = 4
    idle_timeout_s: int = 900
    free_mem_warn_mb: int = 4096
    # 第三视角：多挂一路 320×240 相机，画面里做成第一视角右上角的画中画。
    # 实测默认 15 fps 上限下开与不开都是 14.4 fps（裸渲染 50 vs 55 fps，差异在噪声内）
    pip: bool = True
    # 会话录制：落帧采样率（0 表示不录制），超过最大帧数后丢帧
    record_fps: float = 10.0
    record_max_frames: int = 6000
    # 机器狗网格："" / "none" = 不显示；"go2" = 内置模型；也可给 .object_config.json
    # 或 .glb 的绝对路径。只在画中画里能看到，第一视角视锥看不到它（见 docs/09 #9）
    robot_mesh: str = "go2"
    robot_scale: float = 1.0
    robot_yaw_deg: float = 0.0


@dataclass
class PolicyConfig:
    """模型驾驶：一次 forward 出几个动作、显存门槛、超时。

    全部有默认值——`enabled: false` 时接口直接拒绝，页面也不会显示入口，
    手动驾驶完全不受影响。
    """

    enabled: bool = True
    model_profile: str = ""
    gpu_id: int | None = None
    min_free_mem_mb: int = 12000
    num_frames: int = 8
    num_history: int = 8
    num_future_steps: int = 4
    infer_timeout_s: float = 30.0
    idle_pause_s: float = 60.0


@dataclass
class Settings:
    """进程内全局配置。"""

    root: Path
    server: ServerConfig
    storage: StorageConfig
    scheduler: SchedulerConfig
    security: SecurityConfig
    logging: LoggingConfig
    demo: DemoConfig
    imports: ImportConfig = field(default_factory=ImportConfig)
    sim: SimConfig = field(default_factory=SimConfig)
    policy: PolicyConfig = field(default_factory=PolicyConfig)
    metrics: dict[str, Any] = field(default_factory=dict)
    models: dict[str, Any] = field(default_factory=dict)
    raw: dict[str, Any] = field(default_factory=dict)

    # ---- 常用派生路径 ----
    @property
    def runs_dir(self) -> Path:
        return self.storage.work_dir / "runs"

    @property
    def reports_dir(self) -> Path:
        return self.storage.work_dir / "reports"

    @property
    def cache_dir(self) -> Path:
        return self.storage.work_dir / "cache"

    @property
    def logs_dir(self) -> Path:
        return self.storage.work_dir / "logs"

    @property
    def secret_file(self) -> Path:
        return self.storage.work_dir / "secret.key"

    def ensure_dirs(self) -> None:
        for path in (
            self.storage.work_dir,
            self.runs_dir,
            self.reports_dir,
            self.cache_dir,
            self.logs_dir,
            self.storage.db_path.parent,
        ):
            path.mkdir(parents=True, exist_ok=True)


def _deep_merge(base: dict, override: dict) -> dict:
    result = copy.deepcopy(base)
    for key, value in override.items():
        if isinstance(value, dict) and isinstance(result.get(key), dict):
            result[key] = _deep_merge(result[key], value)
        else:
            result[key] = value
    return result


def _resolve(path_value: str | Path) -> Path:
    path = Path(path_value)
    if not path.is_absolute():
        path = (ROOT_DIR / path).resolve()
    return path


def _load_yaml(path: Path) -> dict[str, Any]:
    if not path.exists():
        return {}
    with path.open("r", encoding="utf-8") as handle:
        return yaml.safe_load(handle) or {}


def _load_import_config(data: dict[str, Any]) -> ImportConfig:
    """组装导入配置；未单独配置备份根目录时沿用历史产物根目录。"""
    raw = data.get("import", {}) or {}
    fallback = raw.get("results_root") or data["demo"]["sample_results_dir"]
    sections = [str(item) for item in raw.get("auto_sections", ALL_IMPORT_SECTIONS)]
    sections = [item for item in sections if item in ALL_IMPORT_SECTIONS]
    return ImportConfig(
        results_root=_resolve(fallback),
        backup_root=_resolve(raw.get("backup_root") or fallback),
        auto_enabled=bool(raw.get("auto_enabled", True)),
        auto_interval_s=max(10, int(raw.get("auto_interval_s", 300))),
        auto_sections=sections or list(ALL_IMPORT_SECTIONS),
        auto_scan_episodes=bool(raw.get("auto_scan_episodes", True)),
        scan_limit_mb=max(1, int(raw.get("scan_limit_mb", 32))),
        scan_exclude=[str(item).lower() for item in raw.get("scan_exclude", ["train", "envdrop", "scalevln"])],
        running_grace_s=max(30, int(raw.get("running_grace_s", 600))),
    )


def _load_sim_config(data: dict[str, Any]) -> SimConfig:
    """组装仿真配置；场景根目录留空时由 episode 解析按数据集元数据回退。"""
    raw = data.get("sim", {}) or {}
    scene_root = str(raw.get("scene_root") or "").strip()
    gpu_ids: list[int] = []
    for item in raw.get("gpu_ids") or [0]:
        try:
            gpu_ids.append(int(item))
        except (TypeError, ValueError):
            continue
    return SimConfig(
        scene_root=_resolve(scene_root) if scene_root else None,
        gpu_ids=gpu_ids or [0],
        tick_hz=float(raw.get("tick_hz", 4.0)),
        render_hz=float(raw.get("render_hz", 15.0)),
        stream_hz=float(raw.get("stream_hz", 20.0)),
        jpeg_quality=max(40, min(98, int(raw.get("jpeg_quality", 82)))),
        max_clients=max(1, int(raw.get("max_clients", 4))),
        idle_timeout_s=max(0, int(raw.get("idle_timeout_s", 900))),
        free_mem_warn_mb=max(0, int(raw.get("free_mem_warn_mb", 4096))),
        pip=bool(raw.get("pip", True)),
        record_fps=max(0.0, min(30.0, float(raw.get("record_fps", 10.0)))),
        record_max_frames=max(0, int(raw.get("record_max_frames", 6000))),
        robot_mesh=str(raw.get("robot_mesh", "go2") or "").strip(),
        robot_scale=max(0.1, min(5.0, float(raw.get("robot_scale", 1.0)))),
        robot_yaw_deg=float(raw.get("robot_yaw_deg", 0.0)),
    )


def _load_policy_config(data: dict[str, Any]) -> PolicyConfig:
    """组装模型驾驶配置；gpu_id 留空表示自动挑空闲显存最多的一张卡。"""
    raw = data.get("policy", {}) or {}
    gpu_raw = raw.get("gpu_id", None)
    gpu_id: int | None = None
    if gpu_raw is not None and str(gpu_raw).strip() != "":
        try:
            gpu_id = int(gpu_raw)
        except (TypeError, ValueError):
            gpu_id = None
    return PolicyConfig(
        enabled=bool(raw.get("enabled", True)),
        model_profile=str(raw.get("model_profile") or "").strip(),
        gpu_id=gpu_id,
        min_free_mem_mb=max(0, int(raw.get("min_free_mem_mb", 12000))),
        num_frames=max(1, int(raw.get("num_frames", 8))),
        num_history=max(1, int(raw.get("num_history", 8))),
        num_future_steps=max(1, int(raw.get("num_future_steps", 4))),
        infer_timeout_s=float(raw.get("infer_timeout_s", 30)),
        idle_pause_s=float(raw.get("idle_pause_s", 60)),
    )


def load_settings(config_path: str | Path | None = None) -> Settings:
    """读取配置并组装为 Settings。"""
    path = Path(config_path) if config_path else Path(
        os.environ.get("NAVEVAL_CONFIG", CONFIG_DIR / "default.yaml")
    )
    data = _deep_merge(DEFAULT_CONFIG, _load_yaml(path))

    # 环境变量覆盖，便于容器化部署
    if os.environ.get("NAVEVAL_HOST"):
        data["server"]["host"] = os.environ["NAVEVAL_HOST"]
    if os.environ.get("NAVEVAL_PORT"):
        data["server"]["port"] = int(os.environ["NAVEVAL_PORT"])
    if os.environ.get("NAVEVAL_WORK_DIR"):
        data["storage"]["work_dir"] = os.environ["NAVEVAL_WORK_DIR"]
    if os.environ.get("NAVEVAL_IMPORT_ROOT"):
        data["import"]["backup_root"] = os.environ["NAVEVAL_IMPORT_ROOT"]

    storage = data["storage"]
    storage_cfg = StorageConfig(
        work_dir=_resolve(storage["work_dir"]),
        db_path=_resolve(storage["db_path"]),
        artifact_keep_days=int(storage.get("artifact_keep_days", 90)),
    )

"""口令哈希与访问令牌。

口令：PBKDF2-HMAC-SHA256，随机 16 字节盐，迭代次数由配置决定。
令牌：base64url(payload).base64url(hmac_sha256(payload))，无第三方依赖。
密钥首次启动生成并保存到 storage/secret.key，重启后令牌仍有效。
"""

from __future__ import annotations

import base64
import hashlib
import hmac
import json
import os
import secrets
import time
from pathlib import Path

from .response import UNAUTHORIZED, AppError

_SECRET: bytes | None = None
_ITERATIONS = 200000


def init_secret(secret_file: Path) -> bytes:
    """初始化签名密钥，文件不存在则生成。"""
    global _SECRET
    if secret_file.exists():
        _SECRET = secret_file.read_bytes()
    else:
        secret_file.parent.mkdir(parents=True, exist_ok=True)
        _SECRET = secrets.token_bytes(32)
        secret_file.write_bytes(_SECRET)
        try:
            os.chmod(secret_file, 0o600)
        except OSError:
            pass
    return _SECRET


def set_iterations(iterations: int) -> None:
    global _ITERATIONS
    _ITERATIONS = max(10000, int(iterations))


def _secret() -> bytes:
    if _SECRET is None:
        raise RuntimeError("签名密钥尚未初始化")
    return _SECRET


def hash_password(password: str, salt: bytes | None = None) -> tuple[str, str]:
    salt = salt or secrets.token_bytes(16)
    digest = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt, _ITERATIONS)
    return digest.hex(), salt.hex()


def verify_password(password: str, password_hash: str, salt_hex: str) -> bool:
    try:
        salt = bytes.fromhex(salt_hex)
    except ValueError:
        return False
    digest = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt, _ITERATIONS)
    return hmac.compare_digest(digest.hex(), password_hash)


def _b64encode(raw: bytes) -> str:
    return base64.urlsafe_b64encode(raw).rstrip(b"=").decode("ascii")


def _b64decode(text: str) -> bytes:
    padding = "=" * (-len(text) % 4)
    return base64.urlsafe_b64decode(text + padding)


def create_token(user_id: int, username: str, role: str, ttl_hours: int) -> str:
    now = int(time.time())
    payload = {
        "uid": user_id,
        "username": username,
        "role": role,
        "iat": now,
        "exp": now + ttl_hours * 3600,
    }
    body = _b64encode(json.dumps(payload, separators=(",", ":")).encode("utf-8"))
    signature = hmac.new(_secret(), body.encode("ascii"), hashlib.sha256).digest()
    return f"{body}.{_b64encode(signature)}"


def decode_token(token: str) -> dict:
    """校验并解析令牌，失败抛 1002。"""
    try:
        body, signature = token.split(".", 1)
    except ValueError as exc:
        raise AppError(UNAUTHORIZED, "令牌格式错误") from exc

    expected = hmac.new(_secret(), body.encode("ascii"), hashlib.sha256).digest()
    try:
        provided = _b64decode(signature)
    except Exception as exc:  # noqa: BLE001 - 解码失败一律视为非法令牌
        raise AppError(UNAUTHORIZED, "令牌签名非法") from exc
    if not hmac.compare_digest(expected, provided):
        raise AppError(UNAUTHORIZED, "令牌签名校验失败")

    try:
        payload = json.loads(_b64decode(body))
    except Exception as exc:  # noqa: BLE001
        raise AppError(UNAUTHORIZED, "令牌内容损坏") from exc

    if int(payload.get("exp", 0)) < int(time.time()):
        raise AppError(UNAUTHORIZED, "登录已过期，请重新登录")
    return payload

"""进程内事件总线。

任务执行线程 / 协程通过 publish 推送进度、资源采样与日志；
SSE 接口通过 subscribe 获取队列，逐个取出并以 text/event-stream 下发。
单机单进程模型下无需 Redis，保持零外部依赖。
"""

from __future__ import annotations

import asyncio
from collections import defaultdict
from typing import Any


class EventBus:
    def __init__(self, queue_size: int = 256) -> None:
        self._channels: dict[int, set[asyncio.Queue]] = defaultdict(set)
        self._queue_size = queue_size
        self._loop: asyncio.AbstractEventLoop | None = None

    def bind_loop(self, loop: asyncio.AbstractEventLoop) -> None:
        """绑定主事件循环，供工作线程安全投递。"""
        self._loop = loop

    def subscribe(self, task_id: int) -> asyncio.Queue:
        queue: asyncio.Queue = asyncio.Queue(maxsize=self._queue_size)
        self._channels[task_id].add(queue)
        return queue

    def unsubscribe(self, task_id: int, queue: asyncio.Queue) -> None:
        channel = self._channels.get(task_id)
        if channel:
            channel.discard(queue)
            if not channel:
                self._channels.pop(task_id, None)

    def publish(self, task_id: int, event: str, data: dict[str, Any]) -> None:
        message = {"event": event, "data": data}
        for queue in list(self._channels.get(task_id, ())):
            self._put(queue, message)

    def publish_threadsafe(self, task_id: int, event: str, data: dict[str, Any]) -> None:
        """从子线程投递，转交主循环执行。"""
        if self._loop is None or not self._loop.is_running():
            self.publish(task_id, event, data)
            return
        self._loop.call_soon_threadsafe(self.publish, task_id, event, data)

    def _put(self, queue: asyncio.Queue, message: dict[str, Any]) -> None:
        try:
            queue.put_nowait(message)
        except asyncio.QueueFull:
            # 消费端过慢时丢弃最旧事件，保证生产者不被阻塞
            try:
                queue.get_nowait()
            except asyncio.QueueEmpty:
                pass
            try:
                queue.put_nowait(message)
            except asyncio.QueueFull:
                pass


BUS = EventBus()
"""数据库引擎与会话。

建库使用 schema.sql 一次性执行，运行时由 SQLAlchemy 管理连接。
SQLite 开启 WAL 与外键约束，超时 30 秒以吸收并发写。
"""

from __future__ import annotations

import sqlite3

from pathlib import Path
from typing import Iterator

from sqlalchemy import Engine, create_engine, event, text
from sqlalchemy.orm import Session, sessionmaker

SCHEMA_FILE = Path(__file__).with_name("schema.sql")
SCHEMA_VERSION = 1

_engine: Engine | None = None
_session_factory: sessionmaker | None = None


def _schema_applied(db_path: Path) -> bool:
    if not db_path.exists():
        return False
    try:
        with sqlite3.connect(db_path) as conn:
            row = conn.execute(
                "SELECT name FROM sqlite_master WHERE type='table' AND name='schema_version'"
            ).fetchone()
            return row is not None
    except sqlite3.DatabaseError:
        return False


def _apply_schema(db_path: Path) -> None:
    db_path.parent.mkdir(parents=True, exist_ok=True)
    sql = SCHEMA_FILE.read_text(encoding="utf-8")
    with sqlite3.connect(db_path) as conn:
        conn.executescript(sql)
        conn.execute(
            "INSERT OR IGNORE INTO schema_version (version, description) VALUES (?, ?)",
            (SCHEMA_VERSION, "initial schema"),
        )
        conn.commit()


def configure(settings: Settings | None = None) -> Engine:
    """构建引擎（幂等）。"""
    global _engine, _session_factory
    settings = settings or get_settings()
    if _engine is not None:
        return _engine

    db_path = settings.storage.db_path
    db_path.parent.mkdir(parents=True, exist_ok=True)
    engine = create_engine(
        f"sqlite:///{db_path}",
        future=True,
        connect_args={"check_same_thread": False, "timeout": 30},
    )

    @event.listens_for(engine, "connect")
    def _set_pragmas(dbapi_connection, _record):  # noqa: ANN001
        cursor = dbapi_connection.cursor()
        cursor.execute("PRAGMA journal_mode=WAL")
        cursor.execute("PRAGMA foreign_keys=ON")
        cursor.execute("PRAGMA busy_timeout=30000")
        cursor.close()

    _engine = engine
    _session_factory = sessionmaker(bind=engine, autoflush=False, expire_on_commit=False)
    return engine


def get_engine() -> Engine:
    if _engine is None:
        return configure()
    return _engine


def get_session_factory() -> sessionmaker:
    if _session_factory is None:
        configure()
    assert _session_factory is not None
    return _session_factory


@contextmanager
def session_scope() -> Iterator[Session]:
    """供服务层使用的会话上下文，异常自动回滚。"""
    session = get_session_factory()()
    try:
        yield session
        session.commit()
    except Exception:
        session.rollback()
        raise
    finally:
        session.close()


def get_db() -> Iterator[Session]:
    """FastAPI 依赖。"""
    session = get_session_factory()()
    try:
        yield session
        session.commit()
    except Exception:
        session.rollback()
        raise
    finally:
        session.close()


def init_database(settings: Settings | None = None) -> bool:
    """建库并执行迁移，返回是否为首次创建。"""
    settings = settings or get_settings()
    settings.ensure_dirs()
    db_path = settings.storage.db_path
    first_time = not _schema_applied(db_path)
    if first_time:
        _apply_schema(db_path)
    configure(settings)
    _migrate()
    return first_time


def _migrate() -> None:
    """按 schema_version 顺序执行补丁脚本（当前仅初始版本）。"""
    migrations_dir = Path(__file__).with_name("migrations")
    if not migrations_dir.exists():
        return

    with session_scope() as session:
        current = session.execute(text("SELECT COALESCE(MAX(version), 0) FROM schema_version")).scalar()

    for script in sorted(migrations_dir.glob("*.sql")):
        try:
            version = int(script.stem.split("_", 1)[0])
        except ValueError:
            continue
        if version <= current:
            continue
        with sqlite3.connect(get_settings().storage.db_path) as conn:
            conn.executescript(script.read_text(encoding="utf-8"))
            conn.execute(
                "INSERT OR IGNORE INTO schema_version (version, description) VALUES (?, ?)",
                (version, script.stem),
            )
            conn.commit()


def reset_engine() -> None:
    """测试用：释放引擎，下次访问重新构建。"""
    global _engine, _session_factory
    if _engine is not None:
        _engine.dispose()
    _engine = None
    _session_factory = None

"""适配器协议与运行上下文。

适配器只负责"跑一条 episode 并产出标准记录"，不写数据库；
调度器负责持久化、事件推送与状态迁移，便于替换推理后端。
"""

from __future__ import annotations

import threading
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any, Callable, Iterator


@dataclass
class EpisodeRecord:
    episode_key: str
    scene_id: str = ""
    success: bool = False
    spl: float = 0.0
    ne: float = 0.0
    os: float = 0.0
    ndtw: float = 0.0
    tl: float = 0.0
    steps: int = 0
    latencies: list[float] = field(default_factory=list)
    gpu_mem_peak_mb: int = 0
    tokens_peak: int = 0
    ctx_len_peak: int = 0
    failure_reason: str = ""
    frames: list[dict[str, Any]] = field(default_factory=list)

    @property
    def latency_avg_ms(self) -> float:
        if not self.latencies:
            return 0.0
        return round(sum(self.latencies) / len(self.latencies), 2)

    @property
    def latency_p95_ms(self) -> float:
        if not self.latencies:
            return 0.0
        ordered = sorted(self.latencies)
        index = min(len(ordered) - 1, int(len(ordered) * 0.95))
        return round(ordered[index], 2)


@dataclass
class RunContext:
    task_id: int
    task_name: str
    run_dir: Path
    config: dict[str, Any]
    episodes: list[dict[str, Any]]
    model: dict[str, Any]
    dataset: dict[str, Any]
    log: Callable[[str], None]
    progress: Callable[[int, int, str], None]
    sample: Callable[[dict[str, Any]], None]
    cancel_event: threading.Event = field(default_factory=threading.Event)
    pause_event: threading.Event = field(default_factory=threading.Event)
    stopped: bool = False

    @property
    def frames_dir(self) -> Path:
        directory = self.run_dir / "frames"
        directory.mkdir(parents=True, exist_ok=True)
        return directory

    def should_cancel(self) -> bool:
        return self.cancel_event.is_set()

    def wait_if_paused(self) -> None:
        while self.pause_event.is_set() and not self.cancel_event.is_set():
            threading.Event().wait(0.3)


class BaseAdapter:
    name = "base"
    allowed_inputs: set[str] = {"rgb"}

    def __init__(self, ctx: RunContext):
        self.ctx = ctx

    def prepare(self) -> None:
        """校验运行条件，失败时抛出 RuntimeError 并附带可读原因。"""

    def run(self) -> Iterator[EpisodeRecord]:
        raise NotImplementedError

    def stop(self) -> None:
        """终止底层进程（取消时调用），默认无操作。"""

    def finish(self) -> dict[str, Any]:
        return {}


def build_adapter(ctx: RunContext) -> BaseAdapter:
    family = str(ctx.model.get("family", "")).lower()
    if family == "streamvln":
        return StreamVLNAdapter(ctx)
    if family == "mock":
        return MockAdapter(ctx)
    entry = str(ctx.model.get("entry") or "")
    if entry.endswith("MockAdapter"):
        return MockAdapter(ctx)
    raise RuntimeError(f"暂不支持的模型家族：{family or '未配置'}")

"""StreamVLN 真实推理适配器。

调用方式与上游 `scripts/run_eval_kaggle.sh` 一致：
    torchrun --nproc_per_node=1 --master_port=<port> <code>/streamvln/streamvln_eval.py \
        --model_path <weights> --eval_split <split> --output_path <raw> --seed <seed> [extra]

运行期持续 tail `<raw>/result.json`（JSONL，一行一条 episode），
解析为标准记录后交给调度器，因此进度是逐条推进的，而不是等进程结束。
"""

from __future__ import annotations

import json
import os
import shutil
import signal
import subprocess
import threading
import time
from collections import deque
from pathlib import Path
from queue import Empty, Queue
from typing import Any, Iterator

from .base import BaseAdapter, EpisodeRecord

# 回放工作台一次最多展示的帧数（与 replay_service.MAX_FRAMES 对齐）
MAX_REPLAY_FRAMES = 600


def _as_float(value: Any, default: float = 0.0) -> float:
    try:
        return float(value)
    except (TypeError, ValueError):
        return default


def _as_int(value: Any, default: int = 0) -> int:
    try:
        return int(float(value))
    except (TypeError, ValueError):
        return default


def _supports_frame_dump(script: Path) -> bool:
    """上游脚本是否支持 --dump_frames（老版本代码里没有这个开关）。"""
    try:
        return "--dump_frames" in script.read_text(encoding="utf-8", errors="replace")
    except OSError:
        return False


class StreamVLNAdapter(BaseAdapter):
    name = "streamvln"

    def __init__(self, ctx):
        super().__init__(ctx)
        self.proc: subprocess.Popen | None = None
        self.result_path: Path | None = None
        # 逐帧回放元数据目录（评测脚本 --dump_frames 的输出），脚本不支持时为 None
        self.replay_dir: Path | None = None
        self._buffer = ""
        self._offset = 0
        self._seen: set[str] = set()
        self._log_queue: Queue[str] = Queue()
        self._tail: deque[str] = deque(maxlen=40)
        self._reader: threading.Thread | None = None

    # ---------------------------------------------------------------- 准备

    def prepare(self) -> None:
        model = self.ctx.model
        code_path = Path(str(model.get("code_path") or "")).expanduser()
        if not code_path.exists():
            raise RuntimeError(f"代码目录不存在：{code_path}")

        script = code_path / "streamvln" / "streamvln_eval.py"
        if not script.exists():
            raise RuntimeError(f"未找到评测脚本：{script}")

        weights = Path(str(model.get("weights_path") or "")).expanduser()
        if not weights.exists():
            raise RuntimeError(f"权重目录不存在：{weights}")

        launcher = str(model.get("entry") or "torchrun").strip()
        launcher_path = launcher if Path(launcher).is_absolute() else shutil.which(launcher)
        if not launcher_path or not Path(launcher_path).exists():
            raise RuntimeError(f"找不到启动器：{launcher}，请检查模型档案中的入口配置")

        raw_dir = self.ctx.run_dir / "raw"
        raw_dir.mkdir(parents=True, exist_ok=True)
        self.result_path = raw_dir / "result.json"

        # 逐帧回放：上游脚本带 --dump_frames 才开启，老版本代码自动降级为「只有汇总指标」
        self.replay_dir = raw_dir / "frames" if _supports_frame_dump(script) else None
        if self.replay_dir is None:
            self.ctx.log("评测脚本不支持 --dump_frames，本次不产出逐帧回放数据")

        command = self._build_command(str(launcher_path), script, weights, raw_dir)
        self.ctx.log("启动命令：" + " ".join(command))

        env = os.environ.copy()
        config = self.ctx.config
        gpu_ids = str(config.get("gpu_ids") or "").strip()
        env["CUDA_VISIBLE_DEVICES"] = gpu_ids or "0"
        env.setdefault("PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION", "python")
        env.setdefault("PYTORCH_CUDA_ALLOC_CONF", "expandable_segments:True")
        env.setdefault("MAGNUM_LOG", "quiet")
        env.setdefault("HABITAT_SIM_LOG", "quiet")

        self.proc = subprocess.Popen(  # noqa: S603 - 命令由本地配置拼装
            command,
            cwd=str(code_path),
            stdout=subprocess.PIPE,
            stderr=subprocess.STDOUT,
            text=True,
            bufsize=1,
            env=env,
            start_new_session=True,
        )
        self._reader = threading.Thread(target=self._read_output, daemon=True)
        self._reader.start()
        self.ctx.log(f"子进程已启动 pid={self.proc.pid}")

    def _build_command(
        self, launcher: str, script: Path, weights: Path, raw_dir: Path
    ) -> list[str]:
        config = self.ctx.config
        eval_split = str(config.get("eval_split") or "val_unseen")
        seed = int(config.get("seed", 42))
        command = [
            launcher,
            "--nproc_per_node=1",
            f"--master_port={config.get('master_port', 29611)}",
            str(script),
            "--model_path",
            str(weights),
            "--eval_split",
            eval_split,
            "--output_path",
            str(raw_dir),
            "--seed",
            str(seed),
        ]
        if self.replay_dir is not None:
            command.extend(["--dump_frames", str(self.replay_dir)])

        extra = config.get("adapter", {}).get("extra_args") or []
        if extra:
            command.extend([str(item) for item in extra])
            return command

        command.extend(["--num_history", str(int(config.get("history_len", 8)))])
        command.extend(["--num_future_steps", str(int(config.get("num_future_steps", 4)))])
        if config.get("memory_policy") == "spatial_prune":
            prune_ratio = float(config.get("prune_ratio", 0.0))
            command.extend(["--prune_mode", "voxel"])
            if 0 < prune_ratio < 1:
                command.extend(["--prune_keep_ratio", f"{1 - prune_ratio:.3f}"])
        return command

    def _read_output(self) -> None:
        assert self.proc is not None and self.proc.stdout is not None
        for line in self.proc.stdout:
            text = line.rstrip()
            if text:
                self._tail.append(text)
                self._log_queue.put(text)

    # ---------------------------------------------------------------- 执行

    def run(self) -> Iterator[EpisodeRecord]:
        if self.proc is None:
            self.prepare()
        assert self.proc is not None

        idle_rounds = 0
        while True:
            if self.ctx.should_cancel():
                self.stop()
                return

            self._flush_logs()

            produced = False
            for payload in self._read_new_records():
                record = self._to_record(payload)
                if record is None or record.episode_key in self._seen:
                    continue
                self._seen.add(record.episode_key)
                produced = True
                yield record

            if self.proc.poll() is not None:
                idle_rounds += 1
                if not produced and idle_rounds >= 2:
                    break
            else:
                idle_rounds = 0

            time.sleep(0.6)

        self._flush_logs()

        if self.proc.returncode not in (0, None):
            tail = "\n".join(list(self._tail)[-12:])
            if "out of memory" in tail.lower():
                raise RuntimeError("推理进程显存不足（OOM）\n" + tail)
            raise RuntimeError(f"推理进程退出码 {self.proc.returncode}\n{tail}")

    def _flush_logs(self) -> None:
        while True:
            try:
                line = self._log_queue.get_nowait()
            except Empty:
                return
            self.ctx.log(line)

    def _read_new_records(self) -> list[dict[str, Any]]:
        """增量读取结果文件，只解析新增的完整行。"""
        if self.result_path is None or not self.result_path.exists():
            return []

        with self.result_path.open("rb") as handle:
            handle.seek(self._offset)
            data = handle.read()
        if not data:
            return []
        self._offset += len(data)

        text = self._buffer + data.decode("utf-8", errors="replace")
        lines = text.split("\n")
        self._buffer = lines.pop()  # 最后一段可能不完整，留到下次拼接

        payloads: list[dict[str, Any]] = []
        for line in lines:
            line = line.strip()
            if not line:
                continue
            try:
                payloads.append(json.loads(line))
            except json.JSONDecodeError:
                continue
        return payloads

    def _load_frames(self, scene_id: str, episode_key: str) -> list[dict[str, Any]]:
        """读取 `--dump_frames` 产出的逐帧元数据，形状与回放服务一致。"""
        if self.replay_dir is None or not self.replay_dir.is_dir():
            return []

        target = self.replay_dir / f"{scene_id}_{episode_key}.jsonl"
        if not target.is_file():
            # 兼容只按 episode_id 命名的实现
            matches = sorted(self.replay_dir.glob(f"*_{episode_key}.jsonl"))
            if not matches:
                return []
            target = matches[0]

        try:
            text = target.read_text(encoding="utf-8", errors="replace")
        except OSError:
            return []

        frames: list[dict[str, Any]] = []
        for line in text.splitlines():
            line = line.strip()
            if not line:
                continue
            try:
                raw = json.loads(line)
            except json.JSONDecodeError:
                continue
            if not isinstance(raw, dict):
                continue
            pose = raw.get("pose")
            if not isinstance(pose, (list, tuple)) or len(pose) < 3:
                continue
            try:
                point = [round(float(value), 3) for value in pose[:3]]
            except (TypeError, ValueError):
                continue

            before = _as_int(raw.get("tokens_before_prune"))
            after = _as_int(raw.get("tokens_after_prune"), before)
            frames.append(
                {
                    "idx": _as_int(raw.get("idx"), len(frames)),
                    "t": round(_as_float(raw.get("t")), 3),
                    "pose": point,
                    "yaw": round(_as_float(raw.get("yaw")), 4),
                    "action": str(raw.get("action") or "forward"),
                    "latency_ms": round(_as_float(raw.get("latency_ms")), 2),
                    "window_len": _as_int(raw.get("window_len")),
                    "slow_update": bool(raw.get("slow_update")),
                    "tokens_before_prune": before,
                    "tokens_after_prune": after,
                    "ctx_len": _as_int(raw.get("ctx_len"), after),
                }
            )

        frames.sort(key=lambda item: item["idx"])
        return frames[:MAX_REPLAY_FRAMES]

    def _to_record(self, payload: dict[str, Any]) -> EpisodeRecord | None:
        episode_key = payload.get("episode_id")
        if episode_key is None:
            return None

        key = str(episode_key)
        scene_id = str(payload.get("scene_id") or "")
        steps = int(payload.get("steps") or 0)
        elapsed = float(payload.get("elapsed_s") or 0.0)
        success = bool(payload.get("success"))

        frames = self._load_frames(scene_id, key)
        step_latencies = [frame["latency_ms"] for frame in frames if frame["latency_ms"] > 0]
        if step_latencies:
            latencies = step_latencies
        else:
            average = round(elapsed / steps * 1000, 2) if steps else 0.0
            latencies = [average] if average else []

        # 逐帧有真实记账就用逐帧峰值；没有则退回 episode 级均值，绝不放大伪造
        tokens_peak = max((frame["tokens_before_prune"] for frame in frames), default=0)
        if not tokens_peak:
            tokens_peak = int(_as_float(payload.get("mean_memory_tokens_before")))
        ctx_len_peak = max((frame["ctx_len"] for frame in frames), default=0)
        if not ctx_len_peak:
            ctx_len_peak = int(_as_float(payload.get("mean_prompt_tokens"))) or int(
                _as_float(payload.get("mean_memory_tokens_after"))
            )

        reason = ""
        if not success:
            max_steps = int(self.ctx.config.get("max_steps", 500))
            reason = "timeout" if steps >= max_steps else "early_stop"

        return EpisodeRecord(
            episode_key=key,
            scene_id=scene_id,
            success=success,
            spl=float(payload.get("spl") or 0.0),
            ne=float(payload.get("ne") or 0.0),
            os=float(payload.get("os") or 0.0),
            ndtw=float(payload.get("ndtw") or 0.0),
            tl=float(payload.get("tl") or 0.0),
            steps=steps,
            latencies=latencies,
            gpu_mem_peak_mb=int(_as_float(payload.get("peak_gpu_mem_mb"))),
            tokens_peak=tokens_peak,
            ctx_len_peak=ctx_len_peak,
            failure_reason=reason,
            frames=frames,
        )

    def stop(self) -> None:
        if self.proc is None or self.proc.poll() is not None:
            return
        try:
            os.killpg(os.getpgid(self.proc.pid), signal.SIGTERM)
        except (ProcessLookupError, PermissionError):
            return
        try:
            self.proc.wait(timeout=10)
        except subprocess.TimeoutExpired:
            try:
                os.killpg(os.getpgid(self.proc.pid), signal.SIGKILL)
            except (ProcessLookupError, PermissionError):
                pass
"""评测任务调度器。

单进程内的异步调度：队列 + 并发额度 + GPU 互斥，任务体在独立线程中执行，
通过事件总线向 SSE 推送进度、资源采样与日志。每条 episode 完成即落库，
异常退出后可按已完成进度继续。
"""

from __future__ import annotations

import asyncio
import json
import logging
import time
from datetime import datetime
from pathlib import Path
from typing import Any

from sqlalchemy import select, update

from ..adapters import BaseAdapter, EpisodeRecord, RunContext, build_adapter
from ..core.config import Settings, get_settings
from ..core.events import BUS
from ..db.session import session_scope
from ..models import (
    Dataset,
    Episode,
    EvalTask,
    ModelProfile,
    ResourceSample,
    TaskEpisode,
)
from . import model_service, task_service

logger = logging.getLogger(__name__)


class Scheduler:
    def __init__(self, settings: Settings | None = None) -> None:
        self.settings = settings or get_settings()
        self._queue: asyncio.Queue[int] = asyncio.Queue()
        self._semaphore = asyncio.Semaphore(max(1, self.settings.scheduler.max_parallel_tasks))
        self._gpu_locks: dict[str, asyncio.Lock] = {}
        self._runtime: dict[int, dict[str, Any]] = {}
        self._dispatcher: asyncio.Task | None = None
        self._loop: asyncio.AbstractEventLoop | None = None
        self._stopping = False

    # ---------------------------------------------------------------- 生命周期

    async def start(self) -> None:
        self._loop = asyncio.get_running_loop()
        BUS.bind_loop(self._loop)

        with session_scope() as db:
            stale = task_service.mark_stale_tasks(db)
            queued = [
                task.id
                for task in db.scalars(
                    select(EvalTask).where(EvalTask.status == "queued").order_by(EvalTask.id)
                ).all()
            ]
        if stale:
            logger.warning("已将 %s 个中断任务标记为失败", stale)

        for task_id in queued:
            self._queue.put_nowait(task_id)
        if queued:
            logger.info("重新入队 %s 个待执行任务", len(queued))

        self._dispatcher = asyncio.create_task(self._dispatch())

    async def shutdown(self) -> None:
        self._stopping = True
        if self._dispatcher:
            self._dispatcher.cancel()
        for task_id, runtime in list(self._runtime.items()):
            runtime["ctx"].cancel_event.set()
            adapter = runtime.get("adapter")
            if adapter is not None:
                await asyncio.to_thread(adapter.stop)
            self._emit(task_id, "state", {"status": "canceled", "reason": "shutdown"})

    # ---------------------------------------------------------------- 入队

    def submit(self, task_id: int) -> None:
        """任意线程安全地入队。"""
        if self._loop and self._loop.is_running():
            self._loop.call_soon_threadsafe(self._queue.put_nowait, task_id)
        else:
            self._queue.put_nowait(task_id)

    async def _dispatch(self) -> None:
        while not self._stopping:
            task_id = await self._queue.get()
            await self._semaphore.acquire()
            asyncio.create_task(self._run_wrapper(task_id))

    async def _run_wrapper(self, task_id: int) -> None:
        gpu_key = ""
        try:
            with session_scope() as db:
                task = db.get(EvalTask, task_id)
                gpu_key = (task.gpu_ids or "").split(",")[0].strip() if task else ""
        except Exception:  # noqa: BLE001 - 读取失败时按无 GPU 约束处理
            logger.exception("读取任务 GPU 配置失败 task=%s", task_id)

        lock = None
        if gpu_key:
            lock = self._gpu_locks.setdefault(gpu_key, asyncio.Lock())
            await lock.acquire()
        try:
            await asyncio.to_thread(self._run_task, task_id)
        except Exception:  # noqa: BLE001
            logger.exception("任务执行线程异常 task=%s", task_id)
        finally:
            if lock is not None:
                lock.release()
            self._semaphore.release()

    # ---------------------------------------------------------------- 执行

    def _run_task(self, task_id: int) -> None:
        with session_scope() as db:
            task = db.get(EvalTask, task_id)
            if task is None or task.status not in ("queued", "paused"):
                return

            task.status = "running"
            task.started_at = task.started_at or task_service.now_str()
            task.error_msg = ""
            config = json.loads(task.config_json or "{}")

            rows = db.execute(
                select(TaskEpisode, Episode)
                .join(Episode, Episode.id == TaskEpisode.episode_id)
                .where(TaskEpisode.task_id == task.id, TaskEpisode.status == "pending")
                .order_by(Episode.id)
            ).all()

            episodes = [
                {
                    "episode_key": episode.episode_id,
                    "scene_id": episode.scene_id,
                    "instruction": episode.instruction,
                    "traj_len": episode.traj_len,
                    "start_pose": json.loads(episode.start_pose_json or "[]"),
                }
                for _, episode in rows
            ]
            row_ids = {episode.episode_id: row.id for row, episode in rows}

            dataset = db.get(Dataset, task.dataset_id)
            model = db.get(ModelProfile, task.model_profile_id)
            dataset_info = {
                "id": dataset.id,
                "name": dataset.name,
                "root_path": dataset.root_path,
                "kind": dataset.kind,
            }
            model_info = model_service.to_dict(model) if model else {}

        if not episodes:
            self._finalize(task_id, error="没有待执行的 episode")
            return

        run_dir = task_service.task_run_dir(task_id)
        (run_dir / "config.json").write_text(
            json.dumps(config, ensure_ascii=False, indent=2), encoding="utf-8"
        )
        log_path = run_dir / "run.log"

        def append_log(text: str) -> None:
            stamp = datetime.now().strftime("%H:%M:%S")
            line = f"[{stamp}] {text}"
            with log_path.open("a", encoding="utf-8") as handle:
                handle.write(line + "\n")
            self._emit(task_id, "log", {"level": "info", "text": text})

        ctx = RunContext(
            task_id=task_id,
            task_name="",
            run_dir=run_dir,
            config=config,
            episodes=episodes,
            model=model_info,
            dataset=dataset_info,
            log=append_log,
            progress=lambda done, total, key: None,
            sample=lambda payload: None,
        )

        self._runtime[task_id] = {"ctx": ctx, "adapter": None}
        self._emit(task_id, "state", {"status": "running", "total": len(episodes)})
        append_log(f"开始执行，待评测 {len(episodes)} 条 episode")

        adapter: BaseAdapter | None = None
        processed = 0
        started = time.time()
        error: str | None = None

        try:
            adapter = build_adapter(ctx)
            self._runtime[task_id]["adapter"] = adapter
            adapter.prepare()

            allowed_keys = set(row_ids)
            for record in adapter.run():
                if ctx.should_cancel():
                    break

                if record.episode_key not in allowed_keys:
                    append_log(f"忽略切分外的结果：{record.episode_key}")
                    continue

                self._persist(task_id, row_ids[record.episode_key], record, run_dir, ctx)
                processed += 1
                self._emit(
                    task_id,
                    "progress",
                    {
                        "done": processed,
                        "total": len(episodes),
                        "episode": record.episode_key,
                        "success": record.success,
                        "spl": record.spl,
                    },
                )
                self._emit(
                    task_id,
                    "sample",
                    {
                        "step": processed,
                        "gpu_mem_mb": record.gpu_mem_peak_mb,
                        "latency_ms": record.latency_avg_ms,
                        "tokens": record.tokens_peak,
                        "ctx_len": record.ctx_len_peak,
                    },
                )

                if ctx.pause_event.is_set():
                    self._set_status(task_id, "paused")
                    self._emit(task_id, "state", {"status": "paused"})
                    append_log("任务已暂停，等待恢复")
                    while ctx.pause_event.is_set() and not ctx.should_cancel():
                        time.sleep(0.3)
                    if ctx.should_cancel():
                        break
                    self._set_status(task_id, "running")
                    self._emit(task_id, "state", {"status": "running"})
                    append_log("任务已恢复")

            adapter.finish()
        except Exception as exc:  # noqa: BLE001 - 需要把失败原因回传界面
            error = str(exc)
            logger.exception("任务执行失败 task=%s", task_id)
            append_log(f"执行失败：{error}")
        finally:
            if ctx.should_cancel() and adapter is not None:
                try:
                    adapter.stop()
                except Exception:  # noqa: BLE001 - 清理阶段忽略
                    logger.exception("终止适配器失败 task=%s", task_id)
            self._runtime.pop(task_id, None)

        if ctx.should_cancel():
            append_log("任务已取消")
            self._finalize(task_id, canceled=True)
        elif error:
            self._finalize(task_id, error=error)
        else:
            append_log(f"执行完成，共处理 {processed} 条，用时 {time.time() - started:.1f}s")
            self._finalize(task_id)

    def _persist(
        self,
        task_id: int,
        row_id: int,
        record: EpisodeRecord,
        run_dir: Path,
        ctx: RunContext,
    ) -> None:
        frames_path = ""
        if record.frames:
            directory = run_dir / "frames"
            directory.mkdir(parents=True, exist_ok=True)
            target = directory / f"{record.episode_key}.jsonl"
            with target.open("w", encoding="utf-8") as handle:
                for frame in record.frames:
                    handle.write(json.dumps(frame, ensure_ascii=False) + "\n")
            frames_path = str(target)

        with session_scope() as db:
            db.execute(
                update(TaskEpisode)
                .where(TaskEpisode.id == row_id)
                .values(
                    status="success" if record.success else "failed",
                    success=1 if record.success else 0,
                    spl=record.spl,
                    ne=record.ne,
                    os=record.os,
                    ndtw=record.ndtw,
                    tl=record.tl,
                    steps=record.steps,
                    latency_avg_ms=record.latency_avg_ms,
                    latency_p95_ms=record.latency_p95_ms,
                    gpu_mem_peak_mb=record.gpu_mem_peak_mb,
                    tokens_peak=record.tokens_peak,
                    ctx_len_peak=record.ctx_len_peak,
                    failure_reason=record.failure_reason,
                    traj_path=frames_path,
                    updated_at=task_service.now_str(),
                )
            )
            task = db.get(EvalTask, task_id)
            progress = 0
            if task is not None:
                task.progress_done = (task.progress_done or 0) + 1
                progress = task.progress_done

            db.add(
                ResourceSample(
                    task_id=task_id,
                    ts=task_service.now_str(),
                    step=progress,
                    gpu_mem_mb=record.gpu_mem_peak_mb,
                    latency_ms=record.latency_avg_ms,
                    tokens=record.tokens_peak,
                    ctx_len=record.ctx_len_peak,
                )
            )

    def _finalize(self, task_id: int, error: str | None = None, canceled: bool = False) -> None:
        with session_scope() as db:
            task = db.get(EvalTask, task_id)
            if task is None:
                return
            metrics = task_service.aggregate_metrics(db, task)
            if canceled:
                task.status = "canceled"
            elif error:
                task.status = "failed"
                task.error_msg = error[:1000]
            else:
                task.status = "succeeded"
            task.finished_at = task_service.now_str()
            if task.started_at:
                try:
                    start = datetime.strptime(task.started_at, "%Y-%m-%d %H:%M:%S")
                    end = datetime.strptime(task.finished_at, "%Y-%m-%d %H:%M:%S")
                    task.duration_s = round((end - start).total_seconds(), 1)
                except ValueError:
                    task.duration_s = None

        self._emit(task_id, "state", {"status": task.status if task else ""})
        self._emit(task_id, "done", {"status": task.status if task else "", "metrics": metrics})
        logger.info("任务结束 task=%s status=%s metrics=%s", task_id, task.status if task else "-", metrics)

    def _set_status(self, task_id: int, status: str) -> None:
        with session_scope() as db:
            task = db.get(EvalTask, task_id)
            if task is not None:
                task.status = status

    @staticmethod
    def _emit(task_id: int, event: str, data: dict[str, Any]) -> None:
        BUS.publish_threadsafe(task_id, event, data)

    # ---------------------------------------------------------------- 控制

    def set_max_parallel(self, value: int) -> None:
        """运行期调整并发额度，新提交的任务立即使用新的信号量。"""
        limit = max(1, min(int(value), 8))
        self.settings.scheduler.max_parallel_tasks = limit
        self._semaphore = asyncio.Semaphore(limit)
        logger.info("并发额度已调整为 %s", limit)

    async def pause(self, task_id: int) -> str:
        runtime = self._runtime.get(task_id)
        if runtime is None:
            raise RuntimeError("任务当前未在运行，无法暂停")
        runtime["ctx"].pause_event.set()
        return "pausing"

    async def resume(self, task_id: int) -> str:
        runtime = self._runtime.get(task_id)
        if runtime is None:
            with session_scope() as db:
                task = task_service.get_task(db, task_id)
                if task.status != "paused":
                    raise RuntimeError("任务不在暂停状态")
                task.status = "queued"
            self.submit(task_id)
            return "queued"
        runtime["ctx"].pause_event.clear()
        return "running"

    async def cancel(self, task_id: int) -> str:
        with session_scope() as db:
            task = task_service.get_task(db, task_id)
            if task.status in task_service.TERMINAL_STATUS:
                return task.status

        runtime = self._runtime.get(task_id)
        if runtime is None:
            with session_scope() as db:
                task = task_service.get_task(db, task_id)
                task.status = "canceled"
                task.finished_at = task_service.now_str()
            self._emit(task_id, "state", {"status": "canceled"})
            return "canceled"

        runtime["ctx"].cancel_event.set()
        adapter = runtime.get("adapter")
        if adapter is not None:
            await asyncio.to_thread(adapter.stop)
        return "canceling"


_scheduler: Scheduler | None = None


def get_scheduler() -> Scheduler:
    global _scheduler
    if _scheduler is None:
        _scheduler = Scheduler()
    return _scheduler

http.interceptors.response.use(
  (response) => {
    const body = response.data
    if (body && typeof body === 'object' && 'code' in body) {
      if (body.code !== 0) {
        ElMessage.error(body.message || '请求失败')
        return Promise.reject(new Error(body.message))
      }
      return body.data
    }
    return body
  },
  (error) => {
    const status = error.response?.status
    const body = error.response?.data
    const message = body?.message || error.message || '网络异常'
    // 轮询/操控这类高频请求用 silent: true：卡住时不要把页面刷成一墙红条
    const silent = Boolean(error.config?.silent)

    if (status === 401) {
      localStorage.removeItem(TOKEN_KEY)
      localStorage.removeItem(USER_KEY)
      if (!window.location.pathname.startsWith('/login')) {
        window.location.href = '/login'
      }
    } else if (!silent) {
      ElMessage.error(message)
    }
    return Promise.reject(new Error(message))
  },
)

import { defineStore } from 'pinia'
import { api, TOKEN_KEY, USER_KEY } from '@/api'

function readUser() {
  try {
    return JSON.parse(localStorage.getItem(USER_KEY) || 'null')
  } catch {
    return null
  }
}

export const useAuthStore = defineStore('auth', {
  state: () => ({
    token: localStorage.getItem(TOKEN_KEY) || '',
    user: readUser(),
  }),

  getters: {
    isLoggedIn: (state) => Boolean(state.token),
    displayName: (state) => state.user?.display_name || state.user?.username || '未登录',
    role: (state) => state.user?.role || 'viewer',
    isAdmin: (state) => state.user?.role === 'admin',
  },

  actions: {
    async login(form) {
      const data = await api.login(form)
      this.token = data.access_token
      this.user = data.user
      localStorage.setItem(TOKEN_KEY, data.access_token)
      localStorage.setItem(USER_KEY, JSON.stringify(data.user))
      return data.user
    },

    async fetchMe() {
      const user = await api.me()
      this.user = user
      localStorage.setItem(USER_KEY, JSON.stringify(user))
      return user
    },

    async logout() {
      try {
        if (this.token) await api.logout()
      } catch {
        // 注销失败不阻塞前端清理
      }
      this.token = ''
      this.user = null
      localStorage.removeItem(TOKEN_KEY)
      localStorage.removeItem(USER_KEY)
    },
  },
})

export function formatBytes(value) {
  const size = Number(value) || 0
  if (size < 1024) return `${size} B`
  if (size < 1024 * 1024) return `${(size / 1024).toFixed(1)} KB`
  return `${(size / 1024 / 1024).toFixed(2)} MB`
}

export function formatPercent(value, digits = 1) {
  if (value === null || value === undefined || Number.isNaN(Number(value))) return '—'
  return `${(Number(value) * 100).toFixed(digits)}%`
}

export function formatNumber(value, digits = 2) {
  if (value === null || value === undefined || Number.isNaN(Number(value))) return '—'
  return Number(value).toFixed(digits)
}

<script setup>
import { computed, onBeforeUnmount, onMounted, ref, watch } from 'vue'

const props = defineProps({
  frames: { type: Array, default: () => [] },
  index: { type: Number, default: 0 },
  height: { type: Number, default: 74 },
})

const emit = defineEmits(['seek'])

const container = ref(null)
const canvas = ref(null)
let ctx = null
let observer = null
const maxValue = computed(() =>
  Math.max(1, ...props.frames.map((frame) => Number(frame.ctx_len || 0))),
)

function draw() {
  const rect = container.value?.getBoundingClientRect()
  if (!rect || !canvas.value) return
  const dpr = window.devicePixelRatio || 1
  const width = rect.width
  const height = rect.height
  canvas.value.width = Math.max(1, Math.round(width * dpr))
  canvas.value.height = Math.max(1, Math.round(height * dpr))
  ctx = canvas.value.getContext('2d')
  ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
  ctx.clearRect(0, 0, width, height)

  const frames = props.frames
  if (!frames.length) return

  const bottom = height - 5
  const gap = frames.length > 90 ? 0 : 1
  const barWidth = Math.max(1, width / frames.length - gap)
  const scale = (bottom - 6) / maxValue.value

  frames.forEach((frame, i) => {
    const value = Number(frame.ctx_len || 0)
    const barHeight = Math.max(2, value * scale)
    const x = (i * width) / frames.length
    const y = bottom - barHeight
    const active = i === props.index
    if (active) {
      ctx.fillStyle = '#2f6fed'
    } else if (frame.slow_update) {
      ctx.fillStyle = '#f8c56b'
    } else {
      ctx.fillStyle = '#d8e6f9'
    }
    ctx.beginPath()
    ctx.roundRect(x, y, barWidth, barHeight, 2)
    ctx.fill()

    if (frame.tokens_after_prune < frame.tokens_before_prune) {
      ctx.fillStyle = '#ef8a8a'
      ctx.fillRect(x, y - 3, barWidth, 2)
    }
  })

  const [current] = [frames[props.index]]
  if (current) {
    const x = (props.index * width) / frames.length
    ctx.strokeStyle = '#2f6fed'
    ctx.lineWidth = 1.5
    ctx.beginPath()
    ctx.moveTo(x + barWidth / 2, 0)
    ctx.lineTo(x + barWidth / 2, bottom)
    ctx.stroke()
  }
}

function onClick(event) {
  if (!props.frames.length) return
  const rect = canvas.value.getBoundingClientRect()
  const ratio = (event.clientX - rect.left) / rect.width
  const index = Math.min(
    props.frames.length - 1,
    Math.max(0, Math.floor(ratio * props.frames.length)),
  )
  emit('seek', index)
}

onMounted(() => {
  draw()
  observer = new ResizeObserver(draw)
  observer.observe(container.value)
})

onBeforeUnmount(() => {
  if (observer) observer.disconnect()
  ctx = null
})

watch(() => [props.frames, props.index], draw, { deep: true })
</script>
